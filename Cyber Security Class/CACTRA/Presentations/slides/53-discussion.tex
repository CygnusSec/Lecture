\begin{frame}{Discussion: Fairness--Throughput Trade-Off}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Fairness improvement}
\begin{itemize}
    \item Starved or disadvantaged flows gain more channel access.
    \item Per-station and per-flow fairness improve.
    \item Cross-layer control reduces excessive TCP aggressiveness.
\end{itemize}

\column{0.48\textwidth}
\textbf{Throughput trade-off}
\begin{itemize}
    \item In some scenarios, fairness can reduce spatial-reuse advantage.
    \item Total end-to-end throughput may decrease.
    \item Total channel utilization remains relatively good.
\end{itemize}
\end{columns}

\vspace{0.35cm}
\KeyPoint{The paper prioritizes fair sharing over letting advantaged flows dominate the wireless medium.}
\end{frame}
