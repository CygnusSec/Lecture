% Let itemize text inherit the ambient \fontsize set at the top of each frame,
% so per-slide sizing actually controls list text. The shared config sets a
% fixed size=\normalsize on these fonts; \setbeamerfont only merges attributes,
% so we must CLEAR the size with the starred form to make inheritance work.
\setbeamerfont*{itemize/enumerate body}{}
\setbeamerfont*{itemize/enumerate subbody}{}
\setbeamerfont*{itemize/enumerate subsubbody}{}

% Compact list spacing so dense source pages fit on a single slide.

\makeatletter
\patchcmd{\@listi}{\itemsep\parsep}{\itemsep 1.5pt \parsep 0pt}{}{}
\patchcmd{\@listii}{\itemsep\parsep}{\itemsep 1pt \parsep 0pt}{}{}
\makeatother

% Monospace code block for command listings: green, tight leading, wraps long lines.
% #1 = font size, #2 = leading, #3 = body (blank lines separate paragraphs).
\newcommand{\CodeBlock}[3]{%
  \par\nopagebreak
  \begingroup
  \color{CommandGreen}\ttfamily\fontsize{#1}{#2}\selectfont
  \setlength{\parindent}{0pt}\setlength{\parskip}{1pt}%
  \raggedright #3\par
  \endgroup}
