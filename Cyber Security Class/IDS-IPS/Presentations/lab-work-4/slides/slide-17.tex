\begin{frame}[t]{Snort Modes}
  \fontsize{15}{18}\selectfont
  \vspace{0.15cm}
  \begin{itemize}
    \item Three main operational modes
    \begin{itemize}
      \item Sniffer Mode
      \item Packet Logger Mode
      \item NIDS Mode
      \item Inline Mode (IPS)
    \end{itemize}
    \item Operational modes are configured via command line switches
    \begin{itemize}
      \item Snort automatically tries to go into NIDS mode if no command line switches are given, looks for snort.conf configuration file in \textcolor{CommandGreen}{/etc/snort}
    \end{itemize}
  \end{itemize}
\end{frame}
