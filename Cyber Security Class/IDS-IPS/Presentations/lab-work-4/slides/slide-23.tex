\begin{frame}[t]{Snort Rules}
  \fontsize{15}{18}\selectfont
  \vspace{0.1cm}
  \begin{itemize}
    \item Snort rules are extremely flexible and are easy to modify, unlike many commercial NIDS
    \item Sample rule to detect SubSeven trojan:
  \end{itemize}
  \vspace{0.15cm}
  \CodeBlock{12}{15}{%
    alert tcp \$EXTERNAL\_NET any -> \$HOME\_NET 21 (msg:"FTP vsFTPd 2.3.4 backdoor login attempt"; flow:to\_server,established; content:":)"; content:"USER "; offset:0; depth:5; nocase; classtype:attempted-admin; sid:1000002; rev:1;)}
  \vspace{0.2cm}
  \begin{itemize}
    \item Elements before parentheses comprise `rule header'
    \item Elements in parentheses are `rule options'
  \end{itemize}
\end{frame}
