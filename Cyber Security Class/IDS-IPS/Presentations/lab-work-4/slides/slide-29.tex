\begin{frame}[t,allowframebreaks]{Installation}
  \fontsize{14}{17}\selectfont
  Run with root privileges (sudo)
  \begin{itemize}
    \item Build and install from source code:
  \end{itemize}
  Step 6: Create a file to store rules and IP lists.
  \CodeBlock{13}{16}{%
    \$ touch /etc/snort/rules/iplists/black\_list.rules\par
    \$ touch /etc/snort/rules/iplists/white\_list.rules\par
    \$ touch /etc/snort/rules/local.rules\par
    \$ touch /etc/snort/sid-msg.map\par}
  \vspace{0.06cm}
  Step 7: Create log storage directory.
  \CodeBlock{13}{16}{%
    \$ mkdir /var/log/snort\par
    \$ mkdir /var/log/snort/archived\_logs\par}
  \vspace{0.06cm}
  Step 8: Setting up username and setting the permissions
  \CodeBlock{13}{16}{%
    \$ useradd snort\par
    \$ chmod -R 5775 /etc/snort\par
    \$ chmod -R 5775 /var/log/snort\par
    \$ chmod -R 5775 /var/log/snort/archived\_logs\par
    \$ chmod -R 5775 /etc/snort/so\_rules\par
    \$ chmod -R 5775 /usr/local/lib/snort\_dynamicrules\par}
\end{frame}
