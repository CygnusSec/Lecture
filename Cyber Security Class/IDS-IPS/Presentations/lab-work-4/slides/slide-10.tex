\begin{frame}[t]{Packet Capture Layer}
  \fontsize{17}{20.5}\selectfont
  \vspace{0.1cm}
  The Packet Capture Layer is the first step in Snort's architecture, where network traffic is captured from the network interface. It uses the libpcap library to intercept packets and pass them to the subsequent layers for analysis.
  \vspace{0.15cm}
  \begin{itemize}
    \item \textcolor{WarningColor}{Interfaces} with network to capture packets
    \item Uses \textcolor{WarningColor}{libpcap/WinPcap} for raw packet access
  \end{itemize}
\end{frame}
