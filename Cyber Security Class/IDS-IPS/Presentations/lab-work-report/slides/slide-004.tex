\begin{frame}[t]{Report Outline II -- Test Environment}
\BodyText{Create and document an isolated environment for testing.}
\begin{itemize}
  \setlength{\itemsep}{0.23cm}
  \item Identify the attacker, firewall/IDS, and victim systems.
  \item Record network interfaces, IP addresses, and routing.
  \item Verify connectivity before scanning.
  \item Explain why the environment is safe and authorized.
\end{itemize}
\end{frame}

\begin{frame}[t]{Report Outline II -- Vulnerability Scanning}

\end{itemize}
\end{frame}
