\begin{frame}[t]{Report Requirements -- Scope}
\BodyText{Select and analyze a vulnerability in \textbf{Metasploitable2} or another approved vulnerable target.}
\vspace{0.25cm}
\begin{itemize}
  \setlength{\itemsep}{0.25cm}
  \item The target must be part of the authorized laboratory environment.
  \item The report must connect discovery, validation, exploitation, detection, and prevention.
  \item Every result must include evidence and an explanation.
\end{itemize}
\end{frame}

\begin{frame}[t]{Report Requirements -- Workflow}
\begin{enumerate}
  \setlength{\itemsep}{0.23cm}
  \item Discover and assess the vulnerability using \Tool{Nmap NSE} and \Tool{Greenbone/OpenVAS}.
  \item Explain the vulnerability and provide evidence from the scan results.
  \item Exploit it using the \Tool{Metasploit Framework} or another appropriate tool.
  \item Propose and test detection and prevention controls.
\end{enumerate}
\end{frame}
