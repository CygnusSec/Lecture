\begin{frame}[t]{nmap (Network Mapper).}
  \begin{itemize}
    \setlength{\itemsep}{0.05cm}
    \item \BodyText{Nmap provides host discovery, port scanning, service detection, and OS detection.}
    \item \BodyText{To accomplish the above, nmap sends specially crafted packets to targets and then analyzes the responses.}
    \item \BodyText{Nmap is available on Linux, Windows, and macOS.}
    \item \BodyText{Install on Linux: \textcolor{CommandGreen}{\texttt{sudo apt install nmap}}}
  \end{itemize}
