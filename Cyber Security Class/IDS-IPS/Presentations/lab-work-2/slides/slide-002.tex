% Prerequisites
\begin{frame}[t]{Prerequisites}
  \BodyText{Students should be familiar with:}
  \begin{itemize}
    \setlength{\itemsep}{0.14cm}
    \item Virtualization platforms such as VMware Workstation, VMware ESXi, VirtualBox, or Hyper-V
    \item Basic TCP/IP networking concepts
    \item Linux fundamentals and command-line operations
    \item Basic concepts of ARP, ICMP, TCP, UDP, ports, and services
  \end{itemize}
\end{frame}
