\begin{frame}[t]{Single Payloads}
  \begin{itemize}
    \item A single payload performs its authorized action without a separate stage
    \item Examples include running one command or creating a temporary lab user when explicitly authorized
    \item Select the least invasive payload that satisfies the lab objective
    \item Stop after collecting the required proof of access
  \end{itemize}
\end{frame}
