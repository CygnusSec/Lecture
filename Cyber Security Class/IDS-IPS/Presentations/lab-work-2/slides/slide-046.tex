\begin{frame}[t]{nmap: Port Scanning - Specify Ports.}
  \begin{itemize}
    \setlength{\itemsep}{0.12cm}
    \item \BodyText{Other than the default 1000 ports, nmap allows you to specify ports by the "-p" option.}
    \item \BodyText{Scan port 80 only}
      \item \CommandText{sudo nmap -p 80 172.16.1.10}
    \item \BodyText{Scan ports 20-28,80, and 100}
      \item \CommandText{sudo nmap -p 20-28,80,100 172.16.1.10}
  \end{itemize}
\end{frame}
