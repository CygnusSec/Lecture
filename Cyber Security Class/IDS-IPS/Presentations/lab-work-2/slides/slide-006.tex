% Authorization and safety
\begin{frame}[t]{Authorization and Lab Safety}
  \begin{itemize}
    \setlength{\itemsep}{0.14cm}
    \item Verify the target IP and use only assigned, intentionally vulnerable lab systems
    \item Stop after proof of access; preserve evidence for the report
  \end{itemize}
  \KeyPoint{Only test systems for which you have explicit authorization.}
\end{frame}
