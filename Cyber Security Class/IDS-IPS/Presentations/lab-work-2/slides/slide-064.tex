\begin{frame}[t]{nmap: Operating System Detection.}
  \begin{itemize}
    \setlength{\itemsep}{0.12cm}
    \item \BodyText{Nmap maintains the \texttt{nmap-os-db} database, which contains:}
      \item \BodyText{probes for detecting various OSes.}
      \item \BodyText{fingerprints to match responses to a certain OS.}
    \item \BodyText{It is located at /usr/share/nmap/nmap-os-db}
    \item \BodyText{Based on this database, nmap tries to determine the OS of a target.}
  \end{itemize}
\end{frame}
