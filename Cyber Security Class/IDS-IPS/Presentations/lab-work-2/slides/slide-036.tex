\begin{frame}[t]{nmap: Port Scanning - Six Port States.}
  \BodyText{Nmap reports one of six states for each scanned port:}

  \vspace{0.12cm}
  \begin{itemize}
    \setlength{\itemsep}{0.22cm}
    \item \BodyText{\textcolor{CommandGreen}{\bfseries open}: An application accepts TCP connections, UDP datagrams, or SCTP associations on the port.}
    \item \BodyText{\textcolor{CommandGreen}{\bfseries closed}: The host responds to probes, but no application is listening on the port.}
    \item \BodyText{\textcolor{CommandGreen}{\bfseries filtered}: Packet filtering prevents the probes from reaching the port, so Nmap cannot determine whether it is open.}
  \end{itemize}
\end{frame}
