\begin{frame}{Configure Ubuntu's WAN Interface}
  \SlideTag{WAN: ens160}
  \Cmd{ip link set dev ens160 down}\\[0.3cm]
  \Cmd{ip addr add 10.10.1.1/24 dev ens160}\\[0.3cm]
  \Cmd{ip link set dev ens160 up}\\[0.55cm]
  \Cmd{ip route}
\end{frame}

\begin{frame}{Configure Ubuntu's Server Interface}
  \SlideTag{SERVER: ens192}
  \Cmd{ip link set dev ens192 down}\\[0.3cm]
  \Cmd{ip addr add 172.16.1.1/24 dev ens192}\\[0.3cm]
  \Cmd{ip link set dev ens192 up}\\[0.55cm]
  \SmallNote{Verify both connected routes and their source addresses.}
\end{frame}

\begin{frame}[fragile]{Make the Ubuntu Network Configuration Persistent}
  Edit \Cmd{/etc/netplan/00-installer-config.yaml}:
  \begin{verbatim}
network:
  ethernets:
    ens160:
      addresses: [10.10.1.1/24]
      dhcp4: false
  \end{verbatim}
\end{frame}

\begin{frame}[fragile]{Add the Server-Facing Interface to Netplan}
  Continue the \Cmd{ethernets} configuration:
  \begin{verbatim}
    ens192:
      addresses: [172.16.1.1/24]
      dhcp4: false
  version: 2
  \end{verbatim}
  \SmallNote{Reference: \href{https://netplan.readthedocs.io/en/stable/examples/}{Netplan configuration examples}}
\end{frame}

\begin{frame}[fragile]{Enable IPv4 Forwarding Persistently}
  Ensure \Cmd{net.ipv4.ip\_forward} is enabled in \Cmd{/etc/sysctl.conf}:
  \vspace{0.35cm}
  \begin{verbatim}
net.ipv4.ip_forward=1
sudo sysctl -p /etc/sysctl.conf
  \end{verbatim}
  \KeyPoint{The Ubuntu router must forward packets between the WAN and server networks.}
\end{frame}

\begin{frame}{Temporary IPv4 Forwarding}
  Use either command-line method for a temporary setting:
  \vspace{0.35cm}
  \Cmd{sudo sysctl -w net.ipv4.ip\_forward=1}\\[0.35cm]
  or\\[0.35cm]
  \Cmd{P=/proc/sys/net/ipv4/ip\_forward}\\
  \Cmd{sudo echo 1 > \$P}
  \vspace{0.25cm}
\end{frame}

\begin{frame}{Configure Kali and Metasploitable2 Temporarily}
  Replace the placeholders with the host's assigned address and router gateway.
  \vspace{0.25cm}
  \Cmd{ip link set dev eth0 down}\\
  \Cmd{ip addr add x.x.x.x/24 dev eth0}\\
  \Cmd{ip link set dev eth0 up}\\
  \Cmd{ip route add default via x.x.x.1}\\[0.3cm]
  \Cmd{ip route}\quad\SmallNote{Confirm the address, connected route, and default route.}
\end{frame}

\begin{frame}[fragile]{Persist the Kali and Metasploitable2 Configuration}
  Edit a file under \Cmd{/etc/network/interfaces.d/*}:
  \begin{verbatim}
allow-hotplug eth0
iface eth0 static
    address 10.10.1.x/24
    gateway 10.10.1.1
  \end{verbatim}
\end{frame}

\begin{frame}{Apply the Kali and Metasploitable2 Configuration}
  Apply the configuration:
  \vspace{0.35cm}
  \Cmd{sudo systemctl restart networking}\\[0.6cm]
  \SmallNote{Reference: \href{https://wiki.debian.org/NetworkConfiguration}{Debian network configuration}}
\end{frame}

\begin{frame}{Validate Connectivity with Ping}
  \Cmd{ping 10.10.1.1}\quad Client-side gateway\\[0.28cm]
  \Cmd{ping 172.16.1.1}\quad Server-side gateway\\[0.28cm]
  \Cmd{ping 10.10.1.x}\quad Attacker host\\[0.28cm]
  \Cmd{ping 172.16.1.x}\quad Victim host
\end{frame}

\begin{frame}{Create the WAN Docker Network}
  \Cmd{D="docker network create"}\\[0.25cm]
  \Cmd{N=10.10.1.0/24}\\[0.25cm]
  \Cmd{R="--ip-range=\$N"}\\[0.25cm]
  \Cmd{S="--subnet=\$N"}\\[0.25cm]
  \Cmd{G="--gateway=10.10.1.254"}\\[0.25cm]
  \Cmd{\$D \$R \$S \$G WANNetwork}
\end{frame}

\begin{frame}{Create the Server Docker Network}
  \Cmd{D="docker network create"}\\[0.25cm]
  \Cmd{N=172.16.1.0/24}\\[0.25cm]
  \Cmd{R="--ip-range=\$N"}\\[0.25cm]
  \Cmd{S="--subnet=\$N"}\\[0.25cm]
  \Cmd{G="--gateway=172.16.1.254"}\\[0.25cm]
  \Cmd{\$D \$R \$S \$G ServerNetwork}
\end{frame}

\begin{frame}{Start the Attacker Container}
  \Cmd{RUN="docker run -dit --privileged"}\\[0.25cm]
  \Cmd{NET="--network WANNetwork"}\\[0.25cm]
  \Cmd{HOST="--hostname kali -p 9392:9392"}\\[0.25cm]
  \Cmd{IMAGE=\$REG/kali-linux:latest}\\[0.25cm]
  \Cmd{\$RUN \$NET \$HOST --name attacker \$IMAGE}
\end{frame}

\begin{frame}{Start the Router Container}
  \Cmd{RUN="docker run -dit --privileged"}\\[0.25cm]
  \Cmd{NET="--network WANNetwork"}\\[0.25cm]
  \Cmd{HOST="--hostname ubuntu --name router"}\\[0.25cm]
  \Cmd{IMAGE=\$REG/ubuntu-router:latest}\\[0.25cm]
  \Cmd{\$RUN \$NET \$HOST \$IMAGE}\\[0.35cm]
  \Cmd{SN=ServerNetwork}\\[0.25cm]
  \Cmd{docker network connect \$SN router}
\end{frame}

\begin{frame}{Start the Victim Container}
  \Cmd{RUN="docker run -dit --privileged"}\\[0.25cm]
  \Cmd{NET="--network ServerNetwork"}\\[0.25cm]
  \Cmd{HOST="--hostname msfadmin --name victim"}\\[0.25cm]
  \Cmd{IMAGE=\$REG/metasploitable2:latest}\\[0.25cm]
  \Cmd{\$RUN \$NET \$HOST \$IMAGE}\\[0.35cm]
  \Cmd{docker attach <container\_name>}
\end{frame}
