\begin{frame}[t,allowframebreaks]{iptables Summary}
  \fontsize{15}{18}\selectfont
  \begin{itemize}
    \item Using Source and Destination IP Addresses
  \end{itemize}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A INPUT -s 172.24.0.0/16 -j DROP\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A OUTPUT -d 172.24.0.0/16 -j DROP\par}
  \begin{itemize}
    \item Filtering by Interface
  \end{itemize}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A INPUT -s 192.168.9.0/24 -i eth0 -j DROP\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A FORWARD -s 192.168.9.0/24 -i eth0 -j DROP\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A FORWARD -s !192.168.9.0/24 -i eth1 -j DROP\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A OUTPUT -s !192.168.9.0/24 -i eth1 -j DROP\par}
  \begin{itemize}
    \item Performing Stateful Inspection
  \end{itemize}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A INPUT -m state -p tcp --dport 80 \textbackslash{}\par}
  --state \textcolor{WarningColor}{\bfseries NEW},\textcolor{WarningColor}{\bfseries ESTABLISHED},\textcolor{WarningColor}{\bfseries RELATED} -j \textcolor{WarningColor}{\bfseries ACCEPT}\par
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# iptables -A OUTPUT -m state -p tcp --sport 80 \textbackslash{}\par}
  --state \textcolor{WarningColor}{\bfseries ESTABLISHED},\textcolor{WarningColor}{\bfseries RELATED} -j \textcolor{WarningColor}{\bfseries ACCEPT}\par
