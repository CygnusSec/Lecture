\begin{frame}[t,allowframebreaks]{TCP Protocol match options (-p tcp ...)}
  \fontsize{15}{18}\selectfont
  \begin{itemize}
    \item --dport : Sets the destination port for the packet. To
  \end{itemize}
  browse the names and aliases of network services and the port numbers they use, view the /etc/services file.\par
  \begin{itemize}
    \item To specify a specific range of port numbers,
  \end{itemize}
  separate the two numbers with a colon (:), such as - p tcp --dport 3000:3200. The largest acceptable valid range is 0:65535.\par
  \begin{itemize}
    \item Use an exclamation point character (!) after the --
  \end{itemize}
  dportoption to tell \textcolor{CommandGreen}{\bfseries iptables} to match all packets which do not use that network service or port.\par
\end{frame}
