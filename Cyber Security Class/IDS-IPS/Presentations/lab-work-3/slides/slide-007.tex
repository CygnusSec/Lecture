\begin{frame}[t,allowframebreaks]{Download -install - start the iptables}
  \fontsize{15}{18}\selectfont
  \begin{itemize}
    \item Update Packages
  \end{itemize}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont sudo apt update\par}
  \begin{itemize}
    \item Install \textcolor{CommandGreen}{\bfseries iptables} and persistence helper
  \end{itemize}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont sudo apt install -y iptables iptables-persistent\par}
  \textcolor{CommandGreen}{\bfseries iptables}-persistent helps keep your rules after reboot.\par
  \begin{itemize}
    \item Verify Installation
  \end{itemize}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont iptables --version\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont ip6tables -version\par}
