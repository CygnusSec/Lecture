\begin{frame}[t,allowframebreaks]{Loading Kernel Modules Needed By iptables}
  \fontsize{15}{18}\selectfont
  \begin{itemize}
    \item The \textcolor{CommandGreen}{\bfseries iptables} application requires you to load certain kernel
  \end{itemize}
  modules to activate some of its functions. Whenever any type of \textcolor{CommandGreen}{\bfseries NAT} is required, the iptable\_\textcolor{CommandGreen}{\bfseries nat} module needs to be loaded. The ip\_conntrack\_ftp module needs to be added for FTP support and should always be loaded with the ip\_conntrack module which tracks TCP connection states. As most scripts probably will keep track of connection states, the ip\_conntrackmodule will be needed in any case. The\par
  ip\_\textcolor{CommandGreen}{\bfseries nat}\_ftp module also needs to be loaded for FTP servers behind a \textcolor{CommandGreen}{\bfseries NAT} firewall.\par
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# File: /etc/rc.local\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# Module to track the state of connections\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont modprobe ip\_conntrack\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# Load the iptables active FTP module, requires ip\_conntrack\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont modprobe ip\_conntrack\_ftp\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# Load iptables NAT module when required\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont modprobe iptable\_nat\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont \# Module required for active an FTP server using NAT\par}
  \vspace{0.08cm}
  {\color{CommandGreen}\ttfamily\fontsize{13.5}{16.5}\selectfont modprobe ip\_nat\_ftp\par}
\end{frame}
