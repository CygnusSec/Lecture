\begin{frame}[t,allowframebreaks]{What Is iptables}
  \fontsize{15}{18}\selectfont
  \begin{itemize}
    \item The built-in chains for the \textcolor{CommandGreen}{\bfseries mangle} table are as
  \end{itemize}
  follows:\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries INPUT}-Alters network packets targeted for the
  \end{itemize}
  host.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries OUTPUT}-Alters locally-generated network
  \end{itemize}
  packets before they are sent out.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries FORWARD}-Alters network packets routed
  \end{itemize}
  through the host.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries PREROUTING}-Alters incoming network
  \end{itemize}
  packets before they are routed.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries POSTROUTING}-Alters network packets before
  \end{itemize}
  they are sent out.\par
\end{frame}
