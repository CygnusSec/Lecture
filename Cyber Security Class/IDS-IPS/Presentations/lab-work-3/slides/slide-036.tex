\begin{frame}[t]{Example of port forwarding}
  \CodeBlock{13}{15}{%
    \# Load the NAT module and Enable routing\par
    modprobe iptable\_nat\par
    echo 1 > /proc/sys/net/ipv4/ip\_forward\par
    external\_int="eth0"\par
    external\_ip="203.132.168.1"\par
    \vspace{0.12cm}
    \# Allow port forwarding for traffic destined to port 80 of the firewall's IP address to be forwarded to port 8080 on server 192.168.1.200. Interface eth0 is the internet interface Interface eth1 is the private network interface\par
    \vspace{0.12cm}
    iptables -t nat -A PREROUTING -p tcp -i eth0 -d \$external\_ip -{}-dport 80 -{}-sport 1024:65535 -j DNAT -{}-to 192.168.1.200:8080\par}
\end{frame}
