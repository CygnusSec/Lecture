\begin{frame}[t,allowframebreaks]{What Is iptables}
  \fontsize{15}{18}\selectfont
  \begin{itemize}
    \item The built-in chains for the \textcolor{CommandGreen}{\bfseries filter} table are as follows:
    \item \textcolor{WarningColor}{\bfseries INPUT}-Applies to network packets that are
  \end{itemize}
  targeted for the host.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries OUTPUT}-Applies to locally-generated network
  \end{itemize}
  packets.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries FORWARD}-Applies to network packets routed
  \end{itemize}
  through the host.\par
  \begin{itemize}
    \item The built-in chains for the \textcolor{CommandGreen}{\bfseries nat} table are as follows:
    \item \textcolor{WarningColor}{\bfseries PREROUTING}-Alters network packets when they
  \end{itemize}
  arrive.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries OUTPUT}-Alters locally-generated network
  \end{itemize}
  packets before they are sent out.\par
  \begin{itemize}
    \item \textcolor{WarningColor}{\bfseries POSTROUTING}-Alters network packets before
  \end{itemize}
  they are sent out.\par
\end{frame}
