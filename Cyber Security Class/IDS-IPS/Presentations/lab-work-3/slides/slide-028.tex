\begin{frame}[t]{Allowing WWW And SSH Access To Firewall}
  \CodeBlock{10}{12.5}{%
    \#-------------------------------------------------\par
    \# Allow previously established connections\par
    \# - Interface eth0 is the internet interface\par
    \#-------------------------------------------------\par
    iptables -A OUTPUT -o eth0 -m state -{}-state ESTABLISHED,RELATED -j ACCEPT\par
    \vspace{0.25cm}
    \#-------------------------------------------------\par
    \# Allow port 80 (www) and 22 (SSH) connections to the firewall\par
    \#-------------------------------------------------\par
    iptables -A INPUT -p tcp -i eth0 -{}-dport 22 -{}-sport 1024:65535 -m state -{}-state NEW -j ACCEPT\par
    \vspace{0.2cm}
    iptables -A INPUT -p tcp -i eth0 -{}-dport 80 -{}-sport 1024:65535 -m state -{}-state NEW -j ACCEPT\par}
\end{frame}
