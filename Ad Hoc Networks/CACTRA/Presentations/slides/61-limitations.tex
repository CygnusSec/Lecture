\begin{frame}{Limitations}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Scope limits}
\begin{itemize}
    \item Mobility-induced unfairness is not considered.
    \item Multi-rate wireless operation is excluded.
    \item Multi-channel operation is excluded.
    \item Evaluation uses IEEE 802.11b and TCP Tahoe.
\end{itemize}

\column{0.48\textwidth}
\textbf{Method limits}
\begin{itemize}
    \item Carrier-sensing flow count is approximated.
    \item $EP$, $Highth$, and $Lowth$ are fixed experimentally.
    \item Sensitivity analysis is not deeply reported.
    \item Exact implementation details are not fully available.
\end{itemize}
\end{columns}
\end{frame}
