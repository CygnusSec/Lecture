\begin{frame}{Research Background}

\KeyPoint{TCP performance and fairness degrade in IEEE 802.11 multi-hop ad hoc networks because all flows share a distributed wireless medium.}

\vspace{0.35cm}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Wireless MAC layer}
\begin{itemize}
    \item IEEE 802.11 DCF
    \item CSMA/CA channel access
    \item Binary Exponential Backoff
    \item Contention Window control
\end{itemize}

\column{0.48\textwidth}
\textbf{Transport layer}
\begin{itemize}
    \item TCP congestion control
    \item Congestion-window growth
    \item Packet injection into shared medium
    \item End-to-end throughput fairness
\end{itemize}
\end{columns}
\end{frame}
