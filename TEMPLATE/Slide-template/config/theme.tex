\newcommand{\FrameAccent}{TitleColor}

\newcommand{\setframeaccent}[1]{%
    \renewcommand{\FrameAccent}{#1}%
    \setbeamercolor{frametitle}{fg=#1,bg=Background}%
    \setbeamercolor{structure}{fg=#1}%
    \setbeamercolor{itemize item}{fg=#1}%
    \setbeamercolor{itemize subitem}{fg=#1}%
    \setbeamercolor{enumerate item}{fg=#1}%
    \setbeamercolor{enumerate subitem}{fg=#1}%
}

\newcommand{\TemplateLogo}{%
    \IfFileExists{images/logo.png}{%
        \begin{tikzpicture}[inner sep=0pt]
            \node (logo) {\includegraphics[width=3.55cm,height=0.82cm,keepaspectratio]{images/logo.png}};
            \node[below=1pt of logo,text=TitleColor,font=\fontsize{6.5}{7.2}\selectfont\bfseries,align=center]
            {Institute of\\Information Technology};
        \end{tikzpicture}%
    }{%
        \begin{tikzpicture}
            \node[draw=LineColor,rounded corners=2pt,inner xsep=6pt,inner ysep=3pt,
                text=SecondaryColor,font=\fontsize{7}{8}\selectfont\bfseries]{YOUR LOGO};
        \end{tikzpicture}%
    }%
}

\setbeamertemplate{background}{%
    \begin{tikzpicture}[remember picture,overlay]
        \IfFileExists{images/slide-background.png}{%
            \node[anchor=center,inner sep=0pt,opacity=0.45]
            at (current page.center) {\includegraphics[width=\paperwidth,height=\paperheight]{images/slide-background.png}};
        }{}%
        \node[anchor=north east,xshift=-0.42cm,yshift=-0.12cm]
        at (current page.north east) {\TemplateLogo};
    \end{tikzpicture}%
}

\setbeamertemplate{frametitle}{%
    \vspace{0.22cm}
    \hspace{0.15cm}%
    \begin{minipage}[t]{0.70\paperwidth}
        \usebeamercolor[fg]{frametitle}%
        \fontsize{17}{20}\selectfont
        \bfseries
        \insertframetitle
    \end{minipage}

    \vspace{0.06cm}

    \hspace{0.15cm}%
    \begin{tikzpicture}
        \draw[draw=\FrameAccent,line width=0.65pt]
        (0,0) -- (10.8,0);
    \end{tikzpicture}

    \vspace{0.02cm}
}

% =========================================================
% Footer
% =========================================================
\setbeamertemplate{footline}{%
    \leavevmode%
    \hbox{%
        \begin{beamercolorbox}[
            wd=.84\paperwidth,
            ht=2.7ex,
            dp=1.2ex,
            left
        ]{author in head/foot}%
            \hspace{0.45cm}%
            \color{SecondaryColor}%
            \fontsize{7}{8.5}\selectfont
            \PresentationFooter
        \end{beamercolorbox}%
        \begin{beamercolorbox}[
            wd=.16\paperwidth,
            ht=2.7ex,
            dp=1.2ex,
            right
        ]{date in head/foot}%
            \color{SecondaryColor}%
            \fontsize{7}{8.5}\selectfont
            \insertframenumber/\inserttotalframenumber\hspace{0.45cm}
        \end{beamercolorbox}%
    }%
}
